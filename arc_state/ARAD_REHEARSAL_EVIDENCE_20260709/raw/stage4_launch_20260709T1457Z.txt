options:
  -h, --help            show this help message and exit
  --harness HARNESS     Controller-seat harness
  --session SESSION     tmux session name
  --window WINDOW       tmux window name
  --resume              attach an existing launcher session
  --dry-run             plan only; no tmux spawn, no provider login
  --preflight           diagnose launch pre-spawn gates without mutating seat,
                        tmux, ledger, or runtime state; exit 0 = all evaluable
                        gates pass and no critical gates skipped; exit 1 = at
                        least one gate WOULD-REFUSE; exit 3 = all evaluable
                        gates pass but one or more critical gates (e.g.
                        containment provisioning) could not be evaluated
                        without a live launch
  --no-tmux             refuse-only flag: request a non-visible/headless seat
                        (always refused)
  --role {controller}   raw role=controller launch request; refuses until a
                        governed takeover evidence packet is supplied
  --takeover-evidence TAKEOVER_EVIDENCE
                        path to a ce takeover --dry-run --json evidence packet
                        that authorizes role=controller launch
  --claude-arg CLAUDE_ARG
                        repeatable extra arg passed to the claude harness (use
                        --claude-arg=<value> for dashed values)
  --codex-arg CODEX_ARG
                        repeatable allowlisted extra arg passed to the codex
                        harness (use --codex-arg=<value> for dashed values)
  --mcp-config MCP_CONFIG
                        CE-owned MCP config path under .ce/state/launch (pins
                        --strict-mcp-config)
  --completion-report-ref COMPLETION_REPORT_REF
                        deterministic completion-report pointer recorded for
                        Ring 0 closeout verification
  --closeout-file CLOSEOUT_FILE
                        deterministic closeout file pointer recorded for Ring
                        0 closeout verification
  --runtime-policy RUNTIME_POLICY
                        v3.5-F: path to the ratified runtime policy whose
                        resource_envelopes bound this seat (systemd-run --user
                        wrap); --dry-run renders the resource_bound block
                        offline
  --backend {gvisor,docker,openshell,local-noop,host}
                        runtime backend selector carried by --runtime-policy
                        (gvisor aliases to gvisor-proxy); host explicitly opts
                        out of contained launch
  --claim-ticket CLAIM_TICKET
                        ce-ops#38: acquire + verify a work-claim lock on this
                        ticket (owner/name#N, an issue URL, or N inside the
                        slug) BEFORE any launch side effect; a foreign active
                        claim refuses the launch
  --repo-root REPO_ROOT
                        repo root for lifecycle registration
  --ledger-root LEDGER_ROOT
                        path to .ce/state/active-work-ledger for lifecycle
                        registration
  --controller-id CONTROLLER_ID
                        owner/controller id recorded in the governed seat
                        lifecycle record
  --host-id HOST_ID     host id recorded in the governed seat lifecycle record
  --purpose PURPOSE     operator-readable purpose recorded in the governed
                        seat lifecycle record
  --json                emit machine-readable JSON
